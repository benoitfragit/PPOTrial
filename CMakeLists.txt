cmake_minimum_required(VERSION 3.15)
project(PPOTrial LANGUAGES CXX)
include(CTest)
option(PPO_BUILD_LEGACY_SIMULATOR "Build the historical Windows simulator" OFF)
option(PPO_BUILD_TORCH "Build the extracted LibTorch backend (provide Torch_DIR)" OFF)
add_subdirectory(RL)
add_subdirectory(Examples/Navigation2D)

if(PPO_BUILD_LEGACY_SIMULATOR)
    if(NOT WIN32)
        message(FATAL_ERROR "Historical simulator dependencies require Windows")
    endif()
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    add_compile_options(/Zc:__cplusplus)
    include(${CMAKE_CURRENT_SOURCE_DIR}/dependencies.cmake)
    if(NOT TARGET ppo_torch)
        add_subdirectory(RL/torch)
    endif()
    add_subdirectory(Serialization)
    add_subdirectory(Entities)
    add_subdirectory(Initializer)
    add_subdirectory(PPO)
    add_subdirectory(Env)
    add_subdirectory(Evaluation)
    add_subdirectory(Renderer)
    add_subdirectory(Agents)
    add_subdirectory(Meta)
    add_subdirectory(Main)
endif()
