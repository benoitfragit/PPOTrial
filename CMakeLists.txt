cmake_minimum_required(VERSION 3.15)
project(PPOTrial LANGUAGES C CXX)
include(CTest)
option(PPO_BUILD_DESKTOP "Build the PPO desktop application" ON)
option(PPO_BUILD_TORCH "Build the LibTorch backend (provide Torch_DIR)" ${PPO_BUILD_DESKTOP})
if(PPO_BUILD_DESKTOP AND NOT PPO_BUILD_TORCH)
    message(FATAL_ERROR "The desktop requires PPO_BUILD_TORCH; use PPO_BUILD_DESKTOP=OFF for headless libraries/tests")
endif()
if(PPO_BUILD_DESKTOP)
    include(${CMAKE_CURRENT_SOURCE_DIR}/dependencies.cmake)
endif()
add_subdirectory(RL)
add_subdirectory(Examples/Navigation2D)
if(PPO_BUILD_DESKTOP)
    add_subdirectory(Main)
    add_subdirectory(Renderer)
endif()
