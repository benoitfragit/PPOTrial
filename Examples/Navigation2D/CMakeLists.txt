add_library(navigation_environment STATIC src/navigation.cpp)
target_include_directories(navigation_environment PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(navigation_environment PUBLIC ppo_environment)
add_executable(navigation2d src/main.cpp)
target_link_libraries(navigation2d PRIVATE navigation_environment)
if(BUILD_TESTING)
    add_executable(navigation_tests tests.cpp)
    target_link_libraries(navigation_tests PRIVATE navigation_environment)
    add_test(NAME navigation_runtime COMMAND navigation_tests)
endif()
if(TARGET ppo_torch)
    add_executable(navigation2d_policy src/torch_main.cpp)
    target_link_libraries(navigation2d_policy PRIVATE navigation_environment ppo_torch)
endif()
if(TARGET ppo_torch)
    add_executable(navigation2d_train src/train_main.cpp)
    target_link_libraries(navigation2d_train PRIVATE navigation_environment ppo_torch)
endif()
option(PPO_NAVIGATION_VIEWER "Build the optional raylib Navigation2D viewer" OFF)
if(PPO_NAVIGATION_VIEWER)
    find_package(raylib CONFIG REQUIRED)
    add_executable(navigation2d_viewer src/viewer.cpp)
    target_link_libraries(navigation2d_viewer PRIVATE navigation_environment raylib)
endif()
